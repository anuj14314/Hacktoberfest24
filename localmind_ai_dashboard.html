Supabase SQL Schema Setup

Copy and run the following SQL script in your Supabase SQL Editor to generate the required persistent tables for LocalMind AI: 

-- Enable UUID Extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Create Messages Persistence Table
CREATE TABLE IF NOT EXISTS public.messages (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    role VARCHAR(20) NOT NULL,
    content TEXT NOT NULL,
    model VARCHAR(50) DEFAULT 'Llama 3',
    citations JSONB DEFAULT '[]'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Create Document Context Table
CREATE TABLE IF NOT EXISTS public.documents (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name TEXT NOT NULL,
    tokens INT DEFAULT 0,
    chunks INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Create Engine User Settings Table
CREATE TABLE IF NOT EXISTS public.user_settings (
    id VARCHAR(50) PRIMARY KEY DEFAULT 'default_config',
    active_model VARCHAR(50) DEFAULT 'Llama 3',
    temperature FLOAT DEFAULT 0.2,
    top_p FLOAT DEFAULT 0.9,
    rep_penalty FLOAT DEFAULT 1.1,
    context_length INT DEFAULT 32768,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Grant Anonymous Public Permissions for Local Desktop App
ALTER TABLE public.messages ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.documents ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.user_settings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public full access to messages" ON public.messages FOR ALL USING (true);
CREATE POLICY "Allow public full access to documents" ON public.documents FOR ALL USING (true);
CREATE POLICY "Allow public full access to user_settings" ON public.user_settings FOR ALL USING (true);
